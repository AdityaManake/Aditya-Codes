void create(){
  struct node *temp, *ptr;
  temp = (struct node*) malloc(sizeof(struct node));
  printf("Enter data\t");
  scanf("%d",&temp->info);
  temp->next = NULL;
  if(start == NULL){
      start = temp;
  }else{
      ptr = start;
      while(ptr->next != NULL){
          ptr = ptr->next;
      }
      ptr->next = temp;
  }
}

void insert_pos(){
  struct node *temp, *ptr;
  int i,pos;
  temp = (struct node*) malloc(sizeof(struct node));
  printf("Enter data\t");
  scanf("%d",&temp->info);
  temp->next = NULL;
  printf("\nEnter position you want to insert to: ");
  scanf("%d",&pos);
  if(start == NULL){
      start = temp;
  }
  else{
    for(i=0, ptr = start ;i<pos-1; i++){
      ptr = ptr->next;
      if(ptr==NULL){
        printf("\npos not found\n");
        return;
      }
    }
    temp->next = ptr->next;
    ptr->next = temp;
  }
}

void delete_all(){
    struct node *temp;
    while(start!=NULL){
    temp = start;
    start = start->next;
    free(temp);
    }
    printf("All Values deleted!");
}

void delete_pos(){
    int i, pos;
    struct node *temp, *ptr;
    if(start == NULL){
        printf("Empty list nothing to delete\n");
    }

    printf("Enter position for element to delete");
    scanf("%d",&pos);

    if(pos==1){
        temp = start;
        start = start->next;
        free(temp);
        return;
    }
    else{
        ptr = start;
        for (i = 1; i < pos - 1; i++){
            if(ptr == NULL){
                printf("Position not found\n");
            }
            ptr = ptr->next;
        }
        temp = ptr->next;
        if(temp == NULL){
            printf("Position not found\n");
        }
        ptr->next = temp->next;
        free(temp);
        printf("Node at given position deleted\n");
    }
}

int count_elems(){
  int count=0;
  struct node *ptr;
  ptr = start;
  while(ptr->next!=NULL){
    ptr = ptr->next;
    count++;
  }
  printf("\nNumber of elements in linked list are: %d",count+1);
  return count+1;
}

void reverse_ll(int countf()){
  int len = countf();
  int arr[len];
  struct node *ptr;
  ptr = start;
  for(int i = len-1; i>=0; i--){
    arr[i] = ptr->info;
    ptr = ptr->next;
  }

  printf("\nReversed Linked List: ");
  for(int i = 0; i<len; i++){
    printf("%d ",arr[i]);
  }
}

void search(int countf()){
  struct node *ptr;
  int key, found=0;

  int count = countf();

  printf("Enter element to search for: ");
  scanf("%d", &key);
  ptr = start;

  for(int i=0; i<count && !found; i++){
    if(key == ptr->info){
      printf("\nThe element was found at position: %d\n",i+1);
      found++;
    }
    ptr = ptr->next;
  }
  if(!found){
    printf("\nElement not in the Linked List\n");
  }
}
